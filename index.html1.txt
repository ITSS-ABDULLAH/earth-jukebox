function playIce(){const d=specs.ice.data,min=Math.min(...d),max=Math.max(...d),now=audio.currentTime,step=.24;d.forEach((v,i)=>{const n=norm(v,min,max);tone(toScale(n),now+i*step,.42,'sine',Math.max(.008,.14*n))})}
function playFires(){const d=specs.fires.data,max=Math.max(...d),now=audio.currentTime,step=.24;d.forEach((v,i)=>{if(v<=0)return;const n=v/max,t=now+i*step;noiseHit(t,.08+n*.08,.1+n*.22);if(n>.5)tone(80+n*40,t,.16,'sine',.25+n*.15,48)})}
function playCo2(){const d=specs.co2.data,min=Math.min(...d),max=Math.max(...d),now=audio.currentTime,step=5.8/d.length;d.forEach((v,i)=>{const n=norm(v,min,max),t=now+i*step,dur=step*1.1;tone(50+n*30,t,dur,'sawtooth',.03+n*.05,52+n*32);tone(80+n*40,t,dur,'sine',.02+n*.03,82+n*42)})}
const playFns={temperature:playTemperature,ice:playIce,fires:playFires,co2:playCo2};

function stop(){runToken++;if(current){setCard(current,false);current=null}if(master&&audio)master.gain.setTargetAtTime(.0001,audio.currentTime,.045);setTimeout(()=>{if(master&&audio&&!current)master.gain.setTargetAtTime(.72,audio.currentTime,.12)},180);document.getElementById('audioAnnouncement').textContent='Playback stopped';resetHero()}
function getDurationMs(id){if(id==='co2')return 6200;const step=.22;return Math.ceil(specs[id].data.length*step*1000)+500}
